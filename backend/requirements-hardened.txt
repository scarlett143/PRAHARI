# Constant-time ML-KEM for deployment.
#
# Kept out of requirements.txt on purpose: liboqs-python builds against the native liboqs
# library, so making it a base dependency would break a plain `pip install -r
# requirements.txt` on any machine without it. The application refuses to start in
# production without this backend, which is the right way round -- a demo install stays
# easy, and a real deployment cannot silently fall back to the pure-Python one.
#
# Needs the native library first:
#   macOS:  brew install liboqs
#   Debian: apt-get install -y cmake ninja-build  (liboqs-python will build liboqs)
-r requirements.txt
liboqs-python==0.16.0
